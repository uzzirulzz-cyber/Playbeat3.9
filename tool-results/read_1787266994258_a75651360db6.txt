    51→  app.get('/api/products', (req, res) => {
    52→    const { category, type, search, sort, status, featured, trending, deal } = req.query;
    53→    let list = [...dbProducts];
    54→
    55→    if (status) {
    56→      list = list.filter(p => p.status === status);
    57→    } else {
    58→      // By default for public queries, return published only unless requested
    59→      if (req.headers['x-admin-query'] !== 'true') {
    60→        list = list.filter(p => p.status === 'published');
    61→      }
    62→    }
    63→
    64→    if (category && category !== 'all') {
    65→      list = list.filter(p => p.categoryId === category || p.slug === category);
    66→    }
    67→
    68→    if (type && type !== 'all') {
    69→      list = list.filter(p => p.productType === type);
    70→    }
    71→
    72→    if (featured === 'true') {
    73→      list = list.filter(p => p.isFeatured);
    74→    }
    75→    if (trending === 'true') {
    76→      list = list.filter(p => p.isTrending);
    77→    }
    78→    if (deal === 'true') {
    79→      list = list.filter(p => p.isFlashDeal);
    80→    }
    81→
    82→    if (search && typeof search === 'string') {
    83→      const q = search.toLowerCase().trim();
    84→      list = list.filter(p =>
    85→        p.title.toLowerCase().includes(q) ||
    86→        p.shortDescription.toLowerCase().includes(q) ||
    87→        p.sku.toLowerCase().includes(q) ||
    88→        p.tags.some(t => t.toLowerCase().includes(q)) ||
    89→        (p.projectorSpecs && (
    90→          p.projectorSpecs.brand.toLowerCase().includes(q) ||
    91→          p.projectorSpecs.model.toLowerCase().includes(q) ||
    92→          p.projectorSpecs.nativeResolution.toLowerCase().includes(q)
    93→        ))
    94→      );
    95→    }
    96→
    97→    if (sort === 'price-low') {
    98→      list.sort((a, b) => a.price - b.price);
    99→    } else if (sort === 'price-high') {
   100→      list.sort((a, b) => b.price - a.price);
   101→    } else if (sort === 'rating') {
   102→      list.sort((a, b) => b.rating - a.rating);
   103→    } else if (sort === 'newest') {
   104→      list.sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime());
   105→    } else {
   106→      // Default: best seller / featured order
   107→      list.sort((a, b) => (b.isFeatured ? 1 : 0) - (a.isFeatured ? 1 : 0));
   108→    }
   109→
   110→    res.json({ products: list, total: list.length });
   111→  });

   223→    const { productIds, action, value } = req.body;

   236→        const factor = 1 + Number(value) / 100;
   237→        return {
   238→          ...p,
   239→          price: Math.max(0.99, Number((p.price * factor).toFixed(2))),
   240→          updatedAt: new Date().toISOString()
   241→        };
   242→      }
   243→      if (action === 'toggle_featured') {
   244→        return { ...p, isFeatured: Boolean(value), updatedAt: new Date().toISOString() };
   245→      }
   246→      if (action === 'toggle_trending') {
   247→        return { ...p, isTrending: Boolean(value), updatedAt: new Date().toISOString() };
   248→      }
   249→      return p;
   250→    });
   251→
   252→    dbAdminLogs.unshift({
   253→      id: `log-${Date.now()}`,
   254→      adminName: 'PlayBeat Admin',
   255→      adminEmail: 'admin@playbeat.digital',
   256→      action: 'Bulk Product Update',
   257→      targetType: 'product',
   258→      details: `Bulk updated ${modifiedCount} products with action "${action}"`,

   759→    const coupon = dbCoupons.find(c => c.code.toUpperCase() === code.toUpperCase() && c.isActive);
   760→    if (!coupon) {
   761→      return res.status(404).json({ valid: false, message: 'Invalid or expired coupon code' });
   762→    }
   763→    if (cartAmount < coupon.minPurchase) {
   764→      return res.status(400).json({ valid: false, message: `Minimum purchase of $${coupon.minPurchase} required` });
   765→    }
   766→    let discount = 0;
   767→    if (coupon.discountType === 'percentage') {
   768→      discount = (cartAmount * coupon.discountValue) / 100;
   769→      if (coupon.maxDiscount) discount = Math.min(coupon.maxDiscount, discount);
   770→    } else {
   771→      discount = coupon.discountValue;
   772→    }
   773→    res.json({ valid: true, coupon, discountAmount: Number(discount.toFixed(2)) });
