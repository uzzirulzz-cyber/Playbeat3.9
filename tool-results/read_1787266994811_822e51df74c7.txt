   239→  const addToast = (type: ToastMessage['type'], title: string, message: string) => {
   240→    const id = `toast-${Date.now()}-${Math.random().toString(36).substring(2, 5)}`;
   241→    setToasts(prev => [...prev, { id, type, title, message }]);

   243→      setToasts(prev => prev.filter(t => t.id !== id));

   247→  const removeToast = (id: string) => {
   248→    setToasts(prev => prev.filter(t => t.id !== id));

   251→  const formatPrice = (pkrPrice: number): string => {
   252→    const safePrice = Number(pkrPrice) || 0;
   253→    const converted = safePrice * currency.rate;
   254→    if (currency.code === 'PKR') {
   255→      return `${currency.symbol}${Math.round(converted).toLocaleString('en-US')}`;
   256→    }
   257→    return `${currency.symbol}${converted.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
   258→  };
   259→
   260→  const addToCart = (product: Product, variation?: ProductVariation, quantity: number = 1) => {
   261→    const activeVar = variation || (product.variations && product.variations.length > 0 ? product.variations[0] : undefined);
   262→    setCart(prev => {
   263→      const existingIdx = prev.findIndex(item => 
   264→        item.productId === product.id && 
   265→        (activeVar ? item.variation?.id === activeVar.id : true)
   266→      );
   267→
   268→      if (existingIdx !== -1) {
   269→        const copy = [...prev];
   270→        copy[existingIdx].quantity += quantity;
   271→        return copy;
   272→      } else {
   273→        return [...prev, { productId: product.id, product, variation: activeVar, quantity }];
   274→      }
   275→    });
   276→
   277→    addToast('success', 'Added to Cart', `${product.title} (${activeVar ? activeVar.value : 'Default'}) added.`);
   278→  };
   279→
   280→  const removeFromCart = (productId: string, variationId?: string) => {
   281→    setCart(prev => prev.filter(item => !(item.productId === productId && (variationId ? item.variation?.id === variationId : true))));
   282→    addToast('info', 'Item Removed', 'Product removed from your cart.');
   283→  };
   284→
   285→  const updateCartQuantity = (productId: string, variationId: string | undefined, qty: number) => {
   286→    if (qty <= 0) {
   287→      removeFromCart(productId, variationId);
   288→      return;
   289→    }
   290→    setCart(prev => prev.map(item => {
   291→      if (item.productId === productId && (variationId ? item.variation?.id === variationId : true)) {
   292→        return { ...item, quantity: qty };
   293→      }
   294→      return item;
   295→    }));
   296→  };
   297→
   298→  const clearCart = () => {
   299→    setCart([]);
   300→  };
   301→
   302→  const cartSubtotal = cart.reduce((sum, item) => {
   303→    const price = item.variation ? item.variation.price : item.product.price;
   304→    return sum + price * item.quantity;
   305→  }, 0);
   306→
   307→  const cartCount = cart.reduce((sum, item) => sum + item.quantity, 0);
   308→
   309→  const toggleWishlist = (productId: string) => {
   310→    setWishlist(prev => {
   311→      const exists = prev.includes(productId);
   312→      if (exists) {
   313→        addToast('info', 'Wishlist Updated', 'Item removed from your wishlist.');
   314→        return prev.filter(id => id !== productId);
   315→      } else {
   316→        addToast('success', 'Wishlist Updated', 'Item saved to your wishlist.');
   317→        return [...prev, productId];
   318→      }
   319→    });
   320→  };
   321→
   322→  const toggleCompare = (product: Product) => {
   323→    setCompareList(prev => {
   324→      const exists = prev.some(p => p.id === product.id);
   325→      if (exists) {
   326→        addToast('info', 'Comparison List', `Removed ${product.title} from comparison.`);
   327→        return prev.filter(p => p.id !== product.id);
   328→      } else {
   329→        if (prev.length >= 4) {
   330→          addToast('warning', 'Comparison Limit', 'You can compare up to 4 projectors at once.');
   331→          return prev;
   332→        }
   333→        addToast('success', 'Added to Compare', `${product.title} added for side-by-side comparison.`);
   334→        return [...prev, product];
   335→      }
   336→    });
   337→  };
   338→
   339→  const clearCompare = () => {
   340→    setCompareList([]);
   341→  };
   342→
   343→  const switchUserRole = (role: User['role']) => {
   344→    setCurrentUser(prev => ({
   345→      ...prev,
   346→      role,
   347→      name: role === 'super_admin' ? 'PlayBeat Super Admin' : role === 'product_manager' ? 'Product Lead' : 'Valued Customer',
   348→      email: role === 'super_admin' ? 'admin@playbeat.digital' : 'customer@playbeat.digital'
   349→    }));
   350→    addToast('info', 'Role Switched', `Active session switched to ${role.replace('_', ' ').toUpperCase()}`);
   351→  };
   352→
   353→  const applyCoupon = (code: string) => {
   354→    const found = coupons.find(c => c.code.toUpperCase() === code.trim().toUpperCase() && (c.isActive !== false && c.status !== 'expired'));
   355→    if (!found) {
   356→      addToast('error', 'Invalid Coupon', 'The promo code entered does not exist or has expired.');
   357→      return { success: false, message: 'Invalid or expired promo code.' };
   358→    }
   359→    const minP = found.minPurchase || found.minOrderAmount || 0;
   360→    if (cartSubtotal < minP) {
   361→      const msg = `Minimum purchase of ${formatPrice(minP)} required for this coupon.`;
   362→      addToast('warning', 'Coupon Requirement', msg);
   363→      return { success: false, message: msg };
   364→    }
   365→    setAppliedCoupon(found);
   366→    addToast('success', 'Promo Code Applied', `Coupon "${found.code}" applied successfully!`);
   367→    return { success: true, message: 'Coupon applied.' };
   368→  };

   370→  const removeCoupon = () => {
   371→    setAppliedCoupon(null);
   372→    addToast('info', 'Coupon Removed', 'Promo discount cleared.');
   373→  };
   374→
   375→  const addCoupon = (newC: Coupon) => {

   378→

   417→        productType: item.product.productType,
   418→        productImage: item.product.images[0],

   420→        variationTitle: item.variation?.value,
   421→        variation: item.variation,
   422→        quantity: item.quantity,
   423→        unitPrice: item.variation ? item.variation.price : item.product.price,
   424→        subtotal: (item.variation ? item.variation.price : item.product.price) * item.quantity,
   425→        totalPrice: (item.variation ? item.variation.price : item.product.price) * item.quantity,
   426→        licenseKeys: item.product.productType === 'digital' ? [

   456→      customerNotes: orderData.customerNotes,
   457→      createdAt: new Date().toISOString(),
   458→      updatedAt: new Date().toISOString()
   459→    };
   460→
   461→    setOrders(prev => [newOrder, ...prev]);
   462→    clearCart();

   467→
   468→  const updateOrderShipment = (orderId: string, trackingNumber: string, carrier: string, status: string) => {

   473→          deliveryStatus: status === 'delivered' ? 'delivered' : 'in_transit',
   474→          fulfillmentStatus: 'shipped_physical',
   475→          items: o.items.map(item => {
   476→            if (item.productType === 'physical_projector') {

   478→                ...item,
   479→                trackingNumber,
   480→                shipment: {
   481→                  trackingNumber,
   482→                  carrier,
   483→                  status,
   484→                  estimatedDelivery: '2026-08-25'
   485→                }
   486→              };
   487→            }
   488→            return item;
   489→          })
   490→        };
   491→      }

   507→      body: JSON.stringify(newContent)
   508→    }).catch(() => {});
   509→    addToast('success', 'Storefront Updated', 'Homepage banners and announcements updated.');
   510→  };
   511→
   512→  const updateG2GConnector = (conn: Partial<G2GSupplierConnector>) => {
   513→    setG2GConnector(prev => ({ ...prev, ...conn }));
   514→    fetch('/api/import/g2g-connector', {
   515→      method: 'PUT',
   516→      headers: { 'Content-Type': 'application/json' },
   517→      body: JSON.stringify(conn)
   518→    }).catch(() => {});
   519→    addToast('success', 'G2G Configuration Saved', 'Supplier markup & category sync settings updated.');
   520→  };
   521→
   522→  const updateG2GSettings = (settings: Partial<G2GSettingsState>) => {
   523→    setG2GSettings(prev => ({ ...prev, ...settings }));
   524→    addToast('success', 'Wholesale Settings Saved', 'G2G profit margin and sync frequency updated.');
   525→  };
   526→
   527→  const syncG2GCatalog = async () => {
   528→    try {
   529→      const res = await fetch('/api/import/g2g-sync', { method: 'POST' });
   530→      const data = await res.json();
   531→      if (data.products) {
   532→        setProducts(data.products);
   533→      }
   534→    } catch (e) {
   535→      console.warn('API sync fallback to client simulation');
   536→    }
   537→    setG2GSettings(prev => ({ ...prev, lastSyncedAt: new Date().toISOString() }));
   538→    addToast('success', 'Catalog Synchronized', 'Wholesale products & live stock updated from G2G supplier.');
   539→  };
   540→
   541→  const cleanAllProductVariations = () => {
   542→    setProducts(prev => prev.map(p => {
   543→      if (!p.variations || p.variations.length <= 1) return p;
   544→      const seen = new Set<string>();
   545→      const clean: ProductVariation[] = [];
   546→      for (const v of p.variations) {
   547→        const key = `${v.type.toLowerCase().trim()}_${v.value.toLowerCase().trim()}`;
   548→        if (!seen.has(key)) {
   549→          seen.add(key);
   550→          clean.push(v);
   551→        }
   552→      }
   553→      return { ...p, variations: clean };
   554→    }));
   555→    addToast('success', 'Deduplication Complete', 'All redundant SKU options cleaned across catalog.');
   556→  };
   557→
   558→  const refreshProducts = () => {
   559→    fetch('/api/products')
   560→      .then(res => res.json())
   561→      .then(data => {
   562→        if (data.products) setProducts(data.products);
   563→      })
   564→      .catch(() => {});
   565→  };
   566→
   567→  const addProduct = (newProduct: Product) => {
   568→    setProducts(prev => [newProduct, ...prev]);
   569→    fetch('/api/products', {
   570→      method: 'POST',
   571→      headers: { 'Content-Type': 'application/json' },
   572→      body: JSON.stringify(newProduct)
   573→    }).catch(() => {});
   574→    addToast('success', 'Product Added', `${newProduct.title} added to catalog.`);
   575→  };
   576→
   577→  const updateProduct = (productOrId: Product | string, maybeUpdates?: Partial<Product>) => {
   578→    if (typeof productOrId === 'string') {
   579→      const id = productOrId;
   580→      const updates = maybeUpdates || {};
   581→      setProducts(prev => prev.map(p => p.id === id ? { ...p, ...updates, updatedAt: new Date().toISOString() } : p));
   582→      fetch(`/api/products/${id}`, {
   583→        method: 'PUT',
   584→        headers: { 'Content-Type': 'application/json' },
   585→        body: JSON.stringify(updates)
   586→      }).catch(() => {});
   587→    } else {
   588→      const updatedProduct = productOrId;
   589→      setProducts(prev => prev.map(p => p.id === updatedProduct.id ? updatedProduct : p));
   590→      fetch(`/api/products/${updatedProduct.id}`, {
   591→        method: 'PUT',
   592→        headers: { 'Content-Type': 'application/json' },
   593→        body: JSON.stringify(updatedProduct)
   594→      }).catch(() => {});
   595→    }
   596→    addToast('success', 'Product Updated', 'Changes saved successfully.');
   597→  };
   598→
   599→  const deleteProduct = (id: string) => {
   600→    setProducts(prev => prev.filter(p => p.id !== id));
   601→    fetch(`/api/products/${id}`, { method: 'DELETE' }).catch(() => {});
   602→    addToast('info', 'Product Deleted', 'Item removed from database.');
   603→  };
   604→
   605→  const approveProduct = (id: string) => {
   606→    updateProduct(id, { status: 'published' });
   607→    addToast('success', 'Product Approved', 'Product is now live on the storefront.');
   608→  };
   609→
   610→  const addNavItem = (item: Omit<NavItem, 'id'>) => {
   611→    const newItem: NavItem = {
   612→      ...item,
   613→      id: `nav-${Date.now()}`
   614→    };
   615→    setNavItems(prev => [...prev, newItem]);
   616→    addToast('success', 'Navigation Item Added', `"${item.label}" added to menu.`);
   617→  };
   618→
   619→  const updateNavItem = (id: string, updates: Partial<NavItem>) => {
   620→    setNavItems(prev => prev.map(item => item.id === id ? { ...item, ...updates } : item));
   621→    addToast('success', 'Navigation Updated', 'Menu item saved successfully.');
   622→  };
   623→
   624→  const deleteNavItem = (id: string) => {
   625→    setNavItems(prev => prev.filter(item => item.id !== id));
   626→    addToast('info', 'Navigation Item Removed', 'Menu item deleted.');
   627→  };
   628→
   629→  const reorderNavItems = (newItems: NavItem[]) => {
   630→    setNavItems(newItems.map((item, idx) => ({ ...item, order: idx + 1 })));
   631→  };
   632→
   633→  const toggleNavItemActive = (id: string) => {
   634→    setNavItems(prev => prev.map(item => item.id === id ? { ...item, isActive: !item.isActive } : item));
   635→  };
   636→
   637→  return (
   638→    <StoreContext.Provider
   639→      value={{
   640→        products,
   641→        categories,
   642→        selectedCategory,
   643→        setSelectedCategory,
   644→        searchQuery,
   645→        setSearchQuery,
   646→        productTypeFilter,
   647→        setProductTypeFilter,
   648→        
   649→        navItems,
   650→        setNavItems,
   651→        addNavItem,
   652→        updateNavItem,
   653→        deleteNavItem,
   654→        reorderNavItems,
   655→        toggleNavItemActive,
   656→        activePromoFilter,
   657→        setActivePromoFilter,
   658→        
   659→        activeView,
   660→        setActiveView,
   661→        adminTab,
   662→        setAdminTab,
   663→        selectedProduct,
   664→        setSelectedProduct,
   665→        isCartOpen,
   666→        setIsCartOpen,
   667→        isWishlistOpen,
   668→        setIsWishlistOpen,
   669→        isAuthModalOpen,
   670→        setIsAuthModalOpen,
   671→        isCheckoutOpen,
   672→        setIsCheckoutOpen,
   673→        isCustomerPortalOpen,
   674→        setIsCustomerPortalOpen,
   675→        isWhatsAppModalOpen,
   676→        setIsWhatsAppModalOpen,
   677→        isCompareModalOpen,
   678→        setIsCompareModalOpen,
   679→        
   680→        cart,
   681→        addToCart,
   682→        removeFromCart,
   683→        updateCartQuantity,
   684→        clearCart,
   685→        cartSubtotal,
   686→        cartCount,
   687→        
   688→        wishlist,
   689→        toggleWishlist,
   690→        
   691→        compareList,
   692→        toggleCompare,
   693→        clearCompare,
   694→        
   695→        currentUser,
   696→        switchUserRole,
   697→        setCurrentUser,
   698→        
   699→        orders,
   700→        createOrder,
   701→        updateOrderShipment,
   702→        updateOrderStatus,
   703→        
   704→        content,
   705→        updateContent,
   706→        coupons,
   707→        appliedCoupon,
   708→        applyCoupon,
   709→        removeCoupon,
   710→        addCoupon,
   711→        deleteCoupon,
   712→        
   713→        g2gConnector,
   714→        g2gSettings,
   715→        updateG2GConnector,
   716→        updateG2GSettings,
   717→        syncG2GCatalog,
   718→        importJobs,
   719→        adminLogs,
   720→        refreshProducts,
   721→        addProduct,
   722→        updateProduct,
   723→        deleteProduct,
   724→        approveProduct,
   725→        cleanAllProductVariations,
   726→        
   727→        toasts,
   728→        addToast,
   729→        removeToast,
   730→        
   731→        currency,
   732→        setCurrency,
   733→        formatPrice
   734→      }}
   735→    >
   736→      {children}
   737→    </StoreContext.Provider>
   738→  );
   739→};
   740→
   741→export const useStore = (): StoreContextType => {
   742→  const context = useContext(StoreContext);
   743→  if (!context) {
   744→    throw new Error('useStore must be used within a StoreProvider');
   745→  }
   746→  return context;
   747→};
